erDiagram USERS ||--o{ ORDERS : "places (สั่งซื้อ)" ORDERS ||--|{ ORDER_ITEMS : "contains (ประกอบด้วย)" PRODUCTS ||--o{ ORDER_ITEMS : "included in (ถูกสั่งซื้อ)" USERS { int id PK "รหัสผู้ใช้ (Auto Increment)" varchar username "ชื่อผู้ใช้สำหรับเข้าสู่ระบบ" varchar password "รหัสผ่านที่เข้ารหัสแล้ว" varchar fullname "ชื่อ-นามสกุล" varchar email "อีเมล" varchar phone "เบอร์โทรศัพท์" varchar role "สิทธิ์การใช้งาน (user / admin)" } ORDERS { int id PK "รหัสคำสั่งซื้อ (Auto Increment)" int user_id FK "รหัสผู้สั่งซื้อ (อ้างอิง USERS.id)" int total_amount "ยอดรวมคำสั่งซื้อ" varchar payment_method "ช่องทางการชำระเงิน" varchar status "สถานะคำสั่งซื้อ (เช่น Pending)" } ORDER_ITEMS { int id PK "รหัสรายการสินค้าในออเดอร์" int order_id FK "รหัสคำสั่งซื้อ (อ้างอิง ORDERS.id)" int product_id FK "รหัสสินค้า (อ้างอิง PRODUCTS.id)" int quantity "จำนวนชิ้นที่สั่งซื้อ" int price "ราคาต่อหน่วย ณ เวลาสั่งซื้อ" } PRODUCTS { int id PK "รหัสสินค้า (Auto Increment)" varchar name "ชื่อสินค้า" varchar gender "เพศ/กลุ่มเป้าหมาย (เช่น men, women)" varchar category "หมวดหมู่สินค้า" varchar subcategory "หมวดหมู่ย่อย" int is_clearance "สถานะลดราคาล้างสต็อก (0/1)" int price "ราคาสินค้า" int stock "จำนวนสินค้าคงเหลือ" varchar image "URL / path รูปภาพสินค้า" }
พจนานุกรมข้อมูล (Data Dictionary)
ตาราง users เก็บข้อมูลผู้ใช้งานและผู้ดูแลระบบ
คอลัมน์ ประเภทข้อมูล Null คีย์ ค่าเริ่มต้น คำอธิบาย
idINT(11)NOPKAuto Incrementรหัสผู้ใช้งาน
usernameVARCHAR(50)NO--ชื่อผู้ใช้ล็อกอิน
passwordVARCHAR(255)NO--รหัสผ่าน (Hash)
fullnameVARCHAR(100)NO--ชื่อ - นามสกุล
emailVARCHAR(100)NO--อีเมล
phoneVARCHAR(20)YES-NULLเบอร์โทรศัพท์
roleVARCHAR(20)YES-'user'สิทธิ์ ('user', 'admin')
ตาราง products เก็บข้อมูลสินค้าและสต็อก
คอลัมน์ ประเภทข้อมูล Null คีย์ ค่าเริ่มต้น คำอธิบาย
idINT(11)NOPKAuto Incrementรหัสสินค้า
nameVARCHAR(100)NO--ชื่อสินค้า
genderVARCHAR(20)NO--กลุ่มเป้าหมาย (men, women)
categoryVARCHAR(50)NO--หมวดหมู่หลัก
subcategoryVARCHAR(50)YES-NULLหมวดหมู่ย่อย
is_clearanceINT(11)YES-0สินค้าลดราคา (0/1)
priceINT(11)NO--ราคา (บาท)
stockINT(11)NO--จำนวนสต็อก
imageVARCHAR(255)YES-NULLURL รูปภาพ
ตาราง orders เก็บรายการคำสั่งซื้อหลัก
คอลัมน์ ประเภทข้อมูล Null คีย์ ค่าเริ่มต้น คำอธิบาย
idINT(11)NOPKAuto Incrementรหัสคำสั่งซื้อ
user_idINT(11)NOFK-รหัสผู้สั่ง (users.id)
total_amountINT(11)NO--ยอดรวมคำสั่งซื้อ
payment_methodVARCHAR(50)YES-NULLช่องทางการชำระเงิน
statusVARCHAR(50)YES-'Pending'สถานะออเดอร์
ตาราง order_items เก็บรายการสินค้าในแต่ละคำสั่งซื้อ
คอลัมน์ ประเภทข้อมูล Null คีย์ ค่าเริ่มต้น คำอธิบาย
idINT(11)NOPKAuto Incrementรหัสรายการสินค้า
order_idINT(11)NOFK-รหัสออเดอร์ (orders.id)
product_idINT(11)NOFK-รหัสสินค้า (products.id)
quantityINT(11)NO--จำนวนชิ้น
priceINT(11)NO--ราคาต่อชิ้น ณ เวลาสั่งซื้อ
